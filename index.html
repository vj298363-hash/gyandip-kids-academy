/* GyanDip Kids Academy 40,000 Dynamic Questions 1 Question = 1 Page 30 Seconds Per Question */ const TOTAL_QUESTIONS = 40000; let student = { name: "", mobile: "" }; let selectedSubject = ""; let currentQuestion = 0; let score = 0; let timer = 30; let timerInterval; let selectedAnswer = null; /* -------------------------------- 40,000 QUESTION ENGINE -------------------------------- */ const subjects = [ "Hindi", "English", "Maths", "Science", "GK" ]; function createQuestion(number, subject) { let q; let correct; let options; if (subject === "Maths") { let a = (number * 7) % 100 + 1; let b = (number * 3) % 50 + 1; correct = a + b; options = [ correct, correct + 1, correct - 1, correct + 2 ]; q = `Question ${number}: ${a} + ${b} = ?`; } else if (subject === "English") { let words = [ ["Cat", "बिल्ली"], ["Dog", "कुत्ता"], ["Book", "किताब"], ["Sun", "सूरज"], ["Water", "पानी"] ]; let item = words[number % words.length]; q = `Question ${number}: "${item[0]}" का Hindi meaning क्या है?`; correct = item[1]; options = [ item[1], "पेड़", "घर", "आसमान" ]; } else if (subject === "Hindi") { let data = [ ["कमल", "फूल"], ["गाय", "पशु"], ["आम", "फल"], ["दिल", "अंग"] ]; let item = data[number % data.length]; q = `Question ${number}: "${item[0]}" क्या है?`; correct = item[1]; options = [ item[1], "पक्षी", "रंग", "खेल" ]; } else if (subject === "Science") { q = `Question ${number}: पौधों को बढ़ने के लिए इनमें से किसकी आवश्यकता होती है?`; correct = "सूर्य का प्रकाश"; options = [ "सूर्य का प्रकाश", "पत्थर", "प्लास्टिक", "लोहे" ]; } else { q = `Question ${number}: भारत की राजधानी क्या है?`; correct = "नई दिल्ली"; options = [ "नई दिल्ली", "मुंबई", "जयपुर", "लखनऊ" ]; } return { question: q, options: shuffle(options), answer: correct }; } /* -------------------------------- SHUFFLE -------------------------------- */ function shuffle(array) { let arr = [...array]; for (let i = arr.length - 1; i > 0; i--) { let j = Math.floor(Math.random() * (i + 1)); [arr[i], arr[j]] = [arr[j], arr[i]]; } return arr; } /* -------------------------------- LOGIN -------------------------------- */ function login() { let name = document.getElementById("studentName").value.trim(); let mobile = document.getElementById("studentMobile").value.trim(); if (!name || !mobile) { alert("कृपया नाम और मोबाइल नंबर भरें।"); return; } student.name = name; student.mobile = mobile; showPage("subjectPage"); } /* -------------------------------- SUBJECT -------------------------------- */ function startTest(subject) { selectedSubject = subject; showPage("feePage"); } /* -------------------------------- START TEST -------------------------------- */ function startQuestions() { currentQuestion = 0; score = 0; showPage("questionPage"); loadQuestion(); } /* -------------------------------- QUESTION -------------------------------- */ function loadQuestion() { clearInterval(timerInterval); selectedAnswer = null; let data = createQuestion( currentQuestion + 1, selectedSubject ); document.getElementById("questionNumber").innerText = `Question ${currentQuestion + 1} / ${TOTAL_QUESTIONS}`; document.getElementById("questionText").innerText = data.question; let optionsBox = document.getElementById("options"); optionsBox.innerHTML = ""; data.options.forEach(option => { let button = document.createElement("button"); button.className = "option"; button.innerText = option; button.onclick = () => { selectedAnswer = option; document .querySelectorAll(".option") .forEach(x => x.classList.remove("selected")); button.classList.add("selected"); }; optionsBox.appendChild(button); }); window.currentCorrectAnswer = data.answer; updateProgress(); startTimer(); } /* -------------------------------- TIMER -------------------------------- */ function startTimer() { timer = 30; document.getElementById("timer").innerText = timer; timerInterval = setInterval(() => { timer--; document.getElementById("timer").innerText = timer; if (timer <= 0) { clearInterval(timerInterval); nextQuestion(); } }, 1000); } /* -------------------------------- NEXT -------------------------------- */ function nextQuestion() { clearInterval(timerInterval); if (selectedAnswer === window.currentCorrectAnswer) { score++; } currentQuestion++; if (currentQuestion >= TOTAL_QUESTIONS) { showResult(); } else { loadQuestion(); } } /* -------------------------------- PROGRESS -------------------------------- */ function updateProgress() { let percentage = (currentQuestion / TOTAL_QUESTIONS) * 100; document.getElementById("progressBar").style.width = percentage + "%"; } /* -------------------------------- RESULT -------------------------------- */ function showResult() { clearInterval(timerInterval); showPage("resultPage"); document.getElementById("score").innerText = score; document.getElementById("resultText").innerText = `${student.name}, आपने ${TOTAL_QUESTIONS} में से ${score} questions सही किए।`; } /* -------------------------------- PAGE SWITCH -------------------------------- */ function showPage(id) { document .querySelectorAll(".page") .forEach(page => page.classList.remove("active")); document .getElementById(id) .classList.add("active"); } /* -------------------------------- LIGHT BACKGROUND TONE -------------------------------- */ let audioContext; document.addEventListener("click", () => { if (!audioContext) { audioContext = new (window.AudioContext || window.webkitAudioContext)(); playSoftTone(); } }, { once: true }); function playSoftTone() { if (!audioContext) return; let oscillator = audioContext.createOscillator(); let gain = audioContext.createGain(); oscillator.frequency.value = 520; gain.gain.value = 0.025; oscillator.connect(gain); gain.connect(audioContext.destination); oscillator.start(); oscillator.stop( audioContext.currentTime + 0.15 ); }
